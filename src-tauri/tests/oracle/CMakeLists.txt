cmake_minimum_required(VERSION 3.16)
project(hissabeth_geopter_oracle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(GEOPTER "${CMAKE_CURRENT_SOURCE_DIR}/../../../reference/upstream")
file(READ "${GEOPTER}/geopter/optical/src/CMakeLists.txt" DEFINITION)
string(REGEX MATCH "set\\(OPTICAL_SRCS[^)]*\\)" SOURCE_LIST "${DEFINITION}")
string(REGEX REPLACE "^set\\(OPTICAL_SRCS|\\)$" "" SOURCE_LIST "${SOURCE_LIST}")
string(REPLACE "\n" ";" SOURCE_LINES "${SOURCE_LIST}")
set(CORE_SOURCES "")
foreach(LINE IN LISTS SOURCE_LINES)
    string(STRIP "${LINE}" LINE)
    if(NOT LINE STREQUAL "")
        list(APPEND CORE_SOURCES "${GEOPTER}/geopter/optical/src/${LINE}")
    endif()
endforeach()
add_library(geopter_core STATIC ${CORE_SOURCES})
target_compile_options(geopter_core PRIVATE -include algorithm)
target_include_directories(geopter_core PUBLIC "${GEOPTER}/geopter/optical/include" "${GEOPTER}/3rdparty/json/include" "${GEOPTER}/3rdparty/eigen-3.3.9")
add_executable(geopter_oracle ../geopter_oracle.cpp)
target_link_libraries(geopter_oracle PRIVATE geopter_core)
